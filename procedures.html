Procedures · 10minilv$evprofe

  1. alumno — todos los módulos, CRN y profesores de un alumno
    CALL alumno(226126177);
  2. reg_alumno — registrar alumno en evaluacion y materiasnn: CodigoAl, 'GRUPO', IdMod, CRN
    CALL reg_alumno(326020336, 'BTEA1BV', 299, 266421);
  3. alumnos_grupo — datos generales del grupo y alumnos con Cal y retroalimentación (CRN opcional)
    CALL alumnos_grupo('TPEA1AV,265905');
  4. mods_grupo — módulos y CRN del grupo con profesor y núm. de alumnos
    CALL mods_grupo('TBIO1AM');
  5. crn_grupo — módulos, CRN y profesores únicos del grupo
    CALL crn_grupo('TBIO1AM');
  6. grupos — todos los grupos M | V con totales
    CALL grupos();
  7. alumnos_crn — todos los registros de un CRN
    CALL alumnos_crn(265905);
  8. prof — grupos, módulos y CRN de un profesor con núm. de alumnos
    CALL prof(2963333);
  9. buscarp — buscar profesor por parte del nombre en evaluacion y profesor
    CALL buscarp('lopez');
  10. buscarm — buscar módulo por nombre: CRN en evaluacion y materiasnn
    CALL buscarm('perspectiva');
  11. reasignar_crn — cambiar profesor: grupo, CodigoP actual, CodigoP nuevo, CRN
    CALL reasignar_crn('TPIN2AM,2026589,2250458,265298');
  12. entrar a mysql — consola Bash, con carga local habilitada (paso previo al 11)
    mysql --local-infile=1 -u 10minilv -h 10minilv.mysql.pythonanywhere-services.com -p '10minilv$evprofe'
    — cargar alumnoscrn.csv (sin encabezado: Id,CodigoAl,NombreAl,Grupo,IdMod,CRN,CodigoP)
    TRUNCATE TABLE alumnoscrn; LOAD DATA LOCAL INFILE '/home/10minilv/alumnoscrn.csv' INTO TABLE alumnoscrn CHARACTER SET utf8 FIELDS TERMINATED BY ',' OPTIONALLY ENCLOSED BY '"' LINES TERMINATED BY '\n' (Id, CodigoAl, NombreAl, Grupo, IdMod, CRN, CodigoP); SELECT COUNT(*) FROM alumnoscrn;
  13. importar_alumnoscrn — pasa a evaluacion y materiasnn los alumnos faltantes de alumnoscrn
    CALL importar_alumnoscrn();
  14. respaldar procedures — consola Bash, genera procedures_evprofe_FECHA.sql en /home/10minilv
    mysqldump -u 10minilv -h 10minilv.mysql.pythonanywhere-services.com -p --routines --no-data --no-create-info --skip-triggers --no-create-db --set-gtid-purged=OFF --no-tablespaces '10minilv$evprofe' > procedures_evprofe_$(date +%F).sql
  15. restaurar procedures — consola Bash, cambia la fecha por la del respaldo
    mysql -u 10minilv -h 10minilv.mysql.pythonanywhere-services.com -p '10minilv$evprofe' < procedures_evprofe_2026-10-12.sql

Funciones consultas.sh · consola Bash

  1. cargar consultas.sh — activa las funciones en la consola (una vez por sesión o tras editar el archivo)
    source ~/consultas.sh
  2. evaluacion — reporte general de todas las evaluaciones → evaluacion<calendario>.csv
    evaluacion 2026B
  3. sin_evaluar — CRN con alumnos sin evaluar por profesor → Sin_Evaluar<calendario>.csv
    sin_evaluar 2026B
  4. evaluacionAlumno — evaluaciones de un alumno → evaluacion_alumno_<CodigoAl>.csv
    evaluacionAlumno 226126177
  5. evaluacionGrupo — evaluaciones de un grupo → evaluacion_grupo_<GRUPO>.csv
    evaluacionGrupo TBIO1AM
  6. importar_tabla — importar CSV a cualquier tabla; con "vaciar" borra antes (pide confirmación)
    importar_tabla alumnoscrn.csv alumnoscrn vaciar
  7. importar_tabla + importar_alumnoscrn — flujo completo de alumnos en una línea
    importar_tabla alumnoscrn.csv alumnoscrn vaciar && mysql -t -u 10minilv -h 10minilv.mysql.pythonanywhere-services.com '10minilv$evprofe' -e "CALL importar_alumnoscrn();"
